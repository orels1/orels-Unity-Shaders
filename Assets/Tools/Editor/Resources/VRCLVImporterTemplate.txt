%Properties()
{
    UI_VRCLightVolumesHeader("# VRC Light Volumes", Int) = 0
    [Toggle(VRCLIGHTVOLUMES)]_VRCLightVolumesEnabled("Enable VRC Light Volumes", Int) = 0
    _VRCLightVolumesSurfacePushoff("Surface Pushoff", Range(0, 1)) = 0.08
    UI_VRCLightVolumesSurfacePushoffNote("> Pushes the volume sampling point away from the surface along the normal vector. This is useful for avoiding leaking and other artifacts on thin surfaces\nValues in 0.05-0.15 range are recommended", Int) = 0
}

%ShaderFeatures()
{
    #pragma shader_feature_local_fragment VRCLIGHTVOLUMES

    // Global keyword to enable in the project
    #pragma shader_feature_fragment DISABLE_VRCLIGHTVOLUMES
}

%ShaderDefines()
{
    #define VRCLV_VERSION 3
    #define VRCLV_MIN_SUPPORTED_VERSION 2
    #define VRCLV_MAX_VOLUMES_COUNT 32
    #define VRCLV_MAX_LIGHTS_COUNT 128
    #define VRCLV_MIN_SPECULAR_PERCEPTUAL_ROUGHNESS 0.089
    #define VRCLV_MIN_N_DOT_V 0.0001

    #if !defined(DISABLE_VRCLIGHTVOLUMES) && defined(VRCLIGHTVOLUMES)
        #define INTEGRATE_VRCLIGHTVOLUMES

        // For PBR shaders on non-baked objects - we skip built-in specular and instead simply use the one provided by VRCLV
        #if !defined(LIGHTMAP_ON)
            #define SKIP_NATIVE_SPECULAR
        #endif

        // Toon Specular uses a custom specular calculation function so we need to
        // 1. Tell the Specular Module to skip its built-in specular calculations (forwardbase only, add pass lights are still handled directly)
        // 2. Tell VRCLV to use a custom specular accumulation
        // 3. Tell Toon fragment base to expose its final Mesh and Surface data structs globally
        #if defined(ORL_LIGHTING_MODEL_TOON_V2) && defined(SPECULAR_ON) && defined(UNITY_PASS_FORWARDBASE)
            #define _INTEGRATE_CUSTOMSPECULAR
            #define LV_CUSTOM_SPECULAR_BRDF
            #define EXPOSE_GLOBAL_DATA
        #endif

        #define LV_PI 3.141592653589793f
        #define LV_INV_PI 0.3183098861837907f
        #define LV_PI2 6.283185307179586f
        #define LV_EVSM_POSITIVE_EXPONENT 5.54f
        #define LV_EVSM_NEGATIVE_EXPONENT 5.0f

        #define _INTEGRATE_CUSTOMPROBES
        #define _INTEGRATE_CUSTOMGI_FLEX
    #endif
}

%%VARIABLES

%AdditionalSurfaceData()
{
    #if defined(INTEGRATE_VRCLIGHTVOLUMES)
    float3 VRCLV_L0;
    float3 VRCLV_L1r;
    float3 VRCLV_L1g;
    float3 VRCLV_L1b;
    float3 VRCLV_Specular;
    #endif
}

%%TEXTURES

%CustomProbesSetup("VRCLightVolumesCustomProbesSetup")
{
    void VRCLightVolumesCustomProbesSetup(MeshData d, inout SurfaceData o)
    {
        #if defined(INTEGRATE_VRCLIGHTVOLUMES)
        {
            float3 L0, L1r, L1g, L1b;
            float3 worldNormal = 0;
            #if defined(ORL_LIGHTING_MODEL_TOON_V2)
            worldNormal = o.WorldNormal;
            #elif defined(ORL_LIGHTING_MODEL_UI)
            worldNormal = d.worldNormal;
            #else
            worldNormal = o.Normal;
            #endif

            float3 specular = 0;
            // PBR Specular handling
            #if defined(ORL_LIGHTING_MODEL_PBR)
            {
                LightVolumeSHSpecular(d.worldSpacePosition + d.worldNormal * _VRCLightVolumesSurfacePushoff, L0, L1r, L1g, L1b, specular, o.Albedo, o.Smoothness, o.Metallic, worldNormal, d.worldSpaceViewDir);
                o.VRCLV_Specular = specular;
            }
            // Toon Specular Setup requires extra steps
            #elif defined(ORL_LIGHTING_MODEL_TOON_V2) && defined(SPECULAR_ON)
            {
                // Recalculate tangent and bitangent for proper aniso specular
                d.worldSpaceTangent.xyz = cross(d.bitangent.xyz, o.WorldNormal);
                d.bitangent.xyz = cross(o.WorldNormal, d.worldSpaceTangent.xyz);

                #if defined(EXPOSE_GLOBAL_DATA)
                GLOBAL_MeshData = d;
                #endif

                LightVolumeSHSpecular(d.worldSpacePosition + d.worldNormal * _VRCLightVolumesSurfacePushoff, L0, L1r, L1g, L1b, specular, o.Albedo, 1.0 - o.SpecularRoughness, o.SpecularAlbedoTint, worldNormal, d.worldSpaceViewDir);
                o.VRCLV_Specular = specular;
            }
            // All other models (UI/VFX) do not use special LV Specular
            #else
            {
                LightVolumeSH(d.worldSpacePosition + d.worldNormal * _VRCLightVolumesSurfacePushoff, L0, L1r, L1g, L1b, 0, worldNormal);
            }
            #endif
            o.VRCLV_L0 = L0;
            o.VRCLV_L1r = L1r;
            o.VRCLV_L1g = L1g;
            o.VRCLV_L1b = L1b;

            // Since toon shaders can derive light directions from probes
            // we need to directly modify these values
            #if defined(ORL_LIGHTING_MODEL_TOON_V2)
            unity_SHAr = float4(L1r, L0.r);
            unity_SHAg = float4(L1g, L0.g);
            unity_SHAb = float4(L1b, L0.b);
            #endif
        }
        #endif
    }
}

%CustomProbes("VRCLightVolumesCustomProbes")
{
    void VRCLightVolumesCustomProbes(SurfaceData o, MeshData d, inout half3 indirectDiffuse)
    {
        // Custom Probes
        #if !defined(ORL_LIGHTING_MODEL_UI)

        #if defined(INTEGRATE_VRCLIGHTVOLUMES)
        {
            #if defined(ORL_LIGHTING_MODEL_TOON_V2)
            {
                o.Normal = _IgnoreLightprobeNormal ? float3(0, 0.5, 0) : o.WorldNormal;
            }
            #endif
            indirectDiffuse = LightVolumeEvaluate(o.Normal, o.VRCLV_L0, o.VRCLV_L1r, o.VRCLV_L1g, o.VRCLV_L1b);
        }
        #endif

        #endif
    }
}

%CustomProbesBakedSpecular("VRCLightVolumesCustomProbesBakedSpecular")
{
    void VRCLightVolumesCustomProbesBakedSpecular(SurfaceData o, MeshData d, inout half3 bakedDirectSpecular, inout half3 bakedSpecularColor, inout half3 bakedDominantDirection)
    {
        #if defined(INTEGRATE_VRCLIGHTVOLUMES)

        // Non-PBR shaders currently use their own specular evaluation
        #if !defined(ORL_LIGHTING_MODEL_PBR)

        bakedSpecularColor = o.VRCLV_L0;
        bakedDominantDirection = o.VRCLV_L1r.xyz + o.VRCLV_L1g.xyz + o.VRCLV_L1b.xyz;

        // PBR Shader uses the VRC Light Volume evaluation instead
        #else
        bakedDirectSpecular = o.VRCLV_Specular;
        #endif
        #endif
    }
}

%CustomGI("VRCLightVolumesCustomGI")
{
    void VRCLightVolumesCustomGI(SurfaceData o, MeshData d, float3 f0, inout half3 indirectDiffuse, inout half3 indirectSpecular)
    {
        #if !defined(ORL_LIGHTING_MODEL_UI)

        #if defined(INTEGRATE_VRCLIGHTVOLUMES)
        {
            #if defined(LIGHTMAP_ON)
            {
                float3 L0, L1r, L1g, L1b;
                float3 lvSpecular;

                float3 worldNormal = 0;
                #if defined(ORL_LIGHTING_MODEL_TOON_V2)
                worldNormal = o.WorldNormal;
                #else
                worldNormal = o.Normal;
                #endif

                LightVolumeAdditiveSHSpecular(d.worldSpacePosition + d.worldNormal * _VRCLightVolumesSurfacePushoff, L0, L1r, L1g,L1b, lvSpecular, o.Albedo, o.Smoothness, o.Metallic, worldNormal, d.worldSpaceViewDir);
                indirectDiffuse += LightVolumeEvaluate(o.Normal, L0, L1r, L1g, L1b);

                indirectSpecular += lvSpecular;
            }
            #endif
        }
        #endif

        #endif
    }
}

// Color mode for UI shaders
%Color("VRCLightVolumesColor")
{
    void VRCLightVolumesColor(MeshData d, bool facing, inout half4 FinalColor)
    {
        #if !defined(UNITY_PASS_FORWARDBASE)
        return;
        #endif
        #if !defined(ORL_LIGHTING_MODEL_UI)
        return;
        #endif

        #if defined(INTEGRATE_VRCLIGHTVOLUMES)
        {
            float3 L0, L1r, L1g, L1b;
            float3 worldNormal = d.worldNormal;
            if (!facing) {
                worldNormal = -worldNormal;
            }
            LightVolumeSH(d.worldSpacePosition + worldNormal * _VRCLightVolumesSurfacePushoff, L0, L1r, L1g, L1b, 0, worldNormal);
            half3 indirectDiffuse = LightVolumeEvaluate(worldNormal, L0, L1r, L1g, L1b);
            FinalColor.rgb *= indirectDiffuse;
        }
        #endif

    }
}

%CustomSpecular("VRCLightVolumesCustomSpecularToon")
{
    void VRCLightVolumesCustomSpecularToon(inout half3 finalSpecular, SurfaceData o)
    {
        #if defined(INTEGRATE_VRCLIGHTVOLUMES)
        {
            // Only Toon v2 Uses this at the moment
            #if defined(ORL_LIGHTING_MODEL_TOON_V2)
            {
                finalSpecular += o.VRCLV_Specular;
            }
            #endif
        }
        #endif
    }
}

%PassFunctions()
{
    #if defined(INTEGRATE_VRCLIGHTVOLUMES) && defined(LV_CUSTOM_SPECULAR_BRDF) && defined(ORL_LIGHTING_MODEL_TOON_V2) && defined(SPECULAR_ON) && defined(EXPOSE_GLOBAL_DATA)

    float3 LV_Custom_SpecularBRDFDirection(float3 f0, float roughness, float roughnessSq, float NoV, float3 worldNormal, float3 viewDir, float3 l0, float3 lightDirNormal, float lightSpreadSq) {
        float centerNoL = dot(worldNormal, lightDirNormal);
        float lightSpread = saturate(sqrt(lightSpreadSq));
        float NoL = saturate((centerNoL + lightSpread) * rcp(1.0 + lightSpread));
        float3 specular = float3(0.0, 0.0, 0.0);
        [branch] if (NoL > 0) {
            float3 lightHalfVector = Unity_SafeNormalize(lightDirNormal + viewDir);
            half NoH = saturate(dot(worldNormal, lightHalfVector));
            half lightLoH = dot(lightDirNormal, lightHalfVector);
            half perceptualRoughness = max(0.01, GLOBAL_SurfaceData.SpecularRoughness);
            perceptualRoughness = perceptualRoughness * (1.7 - 0.7 * perceptualRoughness);
            half anisotropy = GLOBAL_SurfaceData.SpecularAnisotropy * saturate(5.0 * perceptualRoughness);

            perceptualRoughness *= perceptualRoughness;
            perceptualRoughness = max(perceptualRoughness, 0.0045);
            float at = max(perceptualRoughness * (1.0 + anisotropy), 0.001);
            float ab = max(perceptualRoughness * (1.0 - anisotropy), 0.001);

            // Calculate f0 in place with Metallic = 0
            half3 F = F_Schlick(lightLoH, 0.16 * 0.5 * 0.5 + lerp(1, GLOBAL_SurfaceData.Albedo, GLOBAL_SurfaceData.SpecularAlbedoTint));

            float D = D_GGX(NoH, perceptualRoughness);
            half V = V_SmithGGXCorrelated(NoV, NoL, perceptualRoughness);
            half3 isotropicSpecular = max(0, D * V * F);

            D = D_GGX_Anisotropic(NoH, lightHalfVector, GLOBAL_MeshData.worldSpaceTangent, GLOBAL_MeshData.bitangent, at, ab);
            half3 anisotropicSpecular = max(0, D * V * F);

            specular = lerp(isotropicSpecular, anisotropicSpecular, saturate(abs(anisotropy * 100)));
            specular = lerp(specular, smoothstep(0.5, 0.51, specular), GLOBAL_SurfaceData.SpecularSharpness);
            specular *= UNITY_PI;
            specular *= l0;
            specular *= GLOBAL_SurfaceData.SpecularIntensity;
            specular *= GLOBAL_SurfaceData.SpecularTint;
            specular = clamp(specular, 0, GLOBAL_SurfaceData.SpecularIntensity);
            specular *= NoL;
        }
        return specular;
    }

    #endif
}
